cmake_minimum_required(VERSION 3.10)
project(omnifleet_formation_planner)
find_package(ament_cmake REQUIRED)
find_package(ompl REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(joint_plan src/joint_plan.cpp)
target_compile_features(joint_plan PRIVATE cxx_std_17)
target_include_directories(joint_plan PRIVATE ${OMPL_INCLUDE_DIRS})
target_link_libraries(joint_plan ${OMPL_LIBRARIES} nlohmann_json::nlohmann_json)
install(TARGETS joint_plan DESTINATION lib/${PROJECT_NAME})
ament_package()
